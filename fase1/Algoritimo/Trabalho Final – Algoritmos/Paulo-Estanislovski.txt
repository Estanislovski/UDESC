#include <stdio.h>
#include <stdlib.h>

int main(){
    int i, tam = 10;
    int n[tam], pares[tam], impares[tam], somapar = 0, agrimpar = 0;

    for(i = 0; i < tam; i++){
        printf("Digite o %i numero inteiro: ", i + 1);
        scanf("%i", &n[i]);

        if(n[i] % 2 == 0){
            pares[i] = n[i];
            impares[i] = 0;
            somapar += n[i];
        }else{
            impares[i] = n[i];
            pares[i] = 0;
            agrimpar ++;
        }
    }
    //pares
    if(somapar == 0){
        printf("Não foram digitados numeros pares.\n");
    }else{
        printf("Os numeros pares digitados foram:\n");
        for(i = 0; i < tam; i++){
            if(pares[i] != 0){
                printf("%i\n", pares[i]);
            }
        }
        printf("A soma dos numeros pares foi: %i.\n", somapar);
    }
    //impares
    if(agrimpar == 0){
        printf("Não foram digitados numeros impares.\n");
    }else{
        printf("Os numeros impares digitados foram:\n");
        for(i = 0; i < tam; i++){
            if(impares[i] != 0){
                printf("%i\n", impares[i]);
            }
        }
        printf("Foram digitados %i numeros impares.\n", agrimpar);
    }
    return 0;
}

#include <stdio.h>

int main(){
    int i, j;
    int a[10][10];
    
    for(i = 0; i < 10; i++){
        for(j = 0; j < 10; j++){
            if(i < j){
                a[i][j] = 2 * i + 7 * j -2;
                //a[i][j] = 0;
            }else if(i == j){
                //a[i][j] = 1;
                a[i][j] = 3 * (i * i) - 1;
            }else{
                //a[i][j] = 0;
                a[i][j] = 4 * (i * i * i) -5 * (j * j) + 1;
            }
        }
    }

    for(i = 0; i < 10; i++){
        for(j = 0; j < 10; j++){
            printf("%4i ", a[i][j]);
        }
        printf("\n");
    }
return 0;
}

#include <stdio.h>
#include <stdlib.h>

int main(){
    int i, tam = 4;
    float notas[tam], Ma = 0.0;

    for(i = 0; i < (tam - 1); i++){
        printf("Digite a %i nota do Aluno: ", i + 1);
        scanf("%f", &notas[i]);
        notas[tam - 1] += notas[i];
    }
    
    Ma = notas[tam - 1];
    notas[tam - 1] = notas[tam - 1] / (tam - 1);
    Ma = (Ma + notas[tam - 1]) / tam;

    if(Ma >= 9.0){
        printf("A nota do Aluno é 'A'\n");
    }else if(Ma >= 7.5){
        printf("A nota do Aluno é 'B'\n");
    }else if(Ma >= 6.0){
        printf("A nota do Aluno é 'C'\n");
    }else if (Ma >= 4.0){
        printf("A nota do Aluno é 'D'\n");
    }else{
        printf("A nota do Aluno é 'E'\n");
    }
    
    return 0;
}
